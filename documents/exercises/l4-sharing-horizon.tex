\begin{exercisechunk}
\item \textbf{Sequence length and the training budget.\suppmark}
\label[exercise]{ex:sharing-horizon}
\exerciseprofile{3}{2}{3}{\exproof\quad\excalculation\quad\exinterpretation}{%
Compare shared, unshared, and intermediate parameter-growth patterns as
sequence length increases at fixed sample count.}
Use the grouped Bernoulli model and smoothed estimator from
\cref{sec:parameter-sharing}. Assume \(p_j\in[a,1-a]\), with
\(0<a<1/2\).
\begin{enumerate}[(a),beginpenalty=10000]
\item Keep \(n\) fixed and suppose \(p_t\in[a,1-a]\).
Use \cref{ex:sharing-rate} to give explicit upper and lower bounds proportional to
\(T/n\) for the unshared model's expected sequence KL.
Compare this growth with the shared model's limit in
\cref{eq:shared-bernoulli}.
\item Use one new parameter for each successive block of lengths
\(1,2,3,\ldots\). Compute \(d(T)\) at complete-block endpoints and
interpret \cref{eq:sharing-rate}. Use the explicit remainder derived in
\cref{ex:sharing-leading} to determine the leading term even though the
smallest group has size one. How can normalized KL improve while sequence KL
grows?
\end{enumerate}

\end{exercisechunk}
