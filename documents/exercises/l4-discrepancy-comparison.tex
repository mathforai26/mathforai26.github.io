\begin{exercisechunk}
\item \textbf{Task discrepancy as a pseudometric.\suppmark}
\label[exercise]{ex:discrepancy-comparison}
\exerciseprofile{2}{2}{2}{\exproof\quad\exconstruction}{%
Prove that a cost-induced discrepancy is a pseudometric and show why it need
not distinguish laws.}
For probability laws $P,Q$ on $\cX$ and a cost $c:\cX\to\mathbb R$, define
\[
 d_c(P,Q)=|\E_P[c]-\E_Q[c]|.
\]
Prove that \(d_c\) is a pseudometric. Give distinct laws with
\(d_c(P,Q)=0\), first for the binary count and then for another cost.

\end{exercisechunk}
