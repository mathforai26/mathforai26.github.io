\begin{exercisechunk}
\item \textbf{The Dobrushin contraction coefficient.}
\label[exercise]{ex:stability}
\exerciseprofile{3}{2}{2}{\exproof}{%
Prove that the Dobrushin coefficient is the optimal total-variation
contraction factor.}
Let \(\widehat K\) be a Markov kernel on a finite state space. Prove
that the smallest \(\rho\ge0\) satisfying
\[
 \tv{\alpha\widehat K-\beta\widehat K}
 \le \rho\,\tv{\alpha-\beta}
\]
for all probability laws \(\alpha,\beta\) is
\[
 \rho(\widehat K)
 =\max_{x,x'}\tv{\widehat K(x,\cdot)-\widehat K(x',\cdot)}.
\]

\begin{hints}
Normalize the positive and negative parts of
\(\alpha-\beta\) to obtain two probability laws.
\end{hints}
\end{exercisechunk}
