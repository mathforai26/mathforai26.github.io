% Shared exercise chunk: l1-rope
\begin{exercisechunk}
\item \textbf{Encoding relative offsets with RoPE.}
\label[exercise]{ex:rope}
\exerciseprofile{2}{2}{2}{\exproof\quad\excalculation\quad\exconstruction}{%
Derive how paired rotations encode relative displacement and construct a query
for a specified offset.}
Let $T\ge2$ be an integer. Every source position uses the same unrotated unit
key $k\in\mathbb R^2$. For each integer offset $1\le d<T$, we seek a
unit-length query $q(d)\in\mathbb R^2$ that gives the requested source $i-d$
the maximum score at every query position $i>d$.

For a real angle $\phi$, define
\[
R_\phi=\begin{pmatrix}\cos\phi&-\sin\phi\\\sin\phi&\cos\phi\end{pmatrix}.
\]
Fix an \emph{angular frequency} $\omega>0$: moving forward one position
increases the rotation angle by $\omega$ radians. RoPE compares the query at
position $i$ with the key at position $j$ using
\[
s_{ij}=(R_{i\omega}q(d))^\top(R_{j\omega}k),\qquad i,j\in[T].
\]
\begin{enumerate}[label=(\roman*),leftmargin=*,beginpenalty=10000]
\item Show that $R_\alpha^\top R_\beta=R_{\beta-\alpha}$ for real angles
$\alpha,\beta$.
\item Find the unique unit vector $q(d)$ such that $s_{i,i-d}=1$ whenever
$i>d$. Derive a formula for $s_{ij}$ and show that no source has a larger
score.
\item Verify that shifting $i$ and $j$ by the same integer amount leaves the
score unchanged whenever both positions remain in $[T]$.
\end{enumerate}
\end{exercisechunk}
