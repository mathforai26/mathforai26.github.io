\begin{exercisechunk}
\item \textbf{The random-target lower bound.}
\label[exercise]{ex:complete-boxes-lower}
\exerciseprofile{3}{3}{3}{\exproof\quad\excalculation}{%
Use an auxiliary random target to prove that every learner has worst-case
risk at least $\P(U)/2$.}
Show that every learner $\cA$ satisfies
\[
 \max_f R(\cA,f)\ge\frac12\P(U).
\]
Include the hidden-bit independence calculation on the event that the test
input is unseen. Use the
corresponding intermediate steps and ``why?'' questions in
\cref{app:detailed-proof} as a guide, and the probability tools in
\cref{app:probability} as needed.
\end{exercisechunk}
