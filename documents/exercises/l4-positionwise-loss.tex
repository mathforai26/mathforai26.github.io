\begin{exercisechunk}
\item \textbf{A missed dependency at the final position.\suppmark}
\label[exercise]{ex:positionwise-loss}
\exerciseprofile{1}{2}{2}{\excalculation}{%
Show that sequence KL divided by sequence length can vanish while final-answer
error stays fixed.}
Let \(T\ge2\), let \(\cX=\{0,1\}^T\), and let
\(P_{\mathrm{par}}\) and \(Q_{\mathrm{coin}}\) be probability mass functions
on \(\cX\). Under \(P_{\mathrm{par}}\), the
first \(T-1\) bits
are independent fair coins and the last is their parity:
\(X_T=(\sum_{t<T}X_t)\bmod2\). Under \(Q_{\mathrm{coin}}\), all
\(T\) bits are independent fair coins. Define the task cost
\(c_{\mathrm{par}}:\cX\to\{0,1\}\) by
\[
 c_{\mathrm{par}}(x_{1:T})
 =\I\!\left\{x_T\ne\Bigl(\sum_{t<T}x_t\Bigr)\bmod2\right\}.
\]
\begin{enumerate}[(a),beginpenalty=10000]
\item Show that for every \(t<T\) and every
\(x_{<t}\in\{0,1\}^{t-1}\),
\[
 P_{\mathrm{par}}(\mathord\cdot\mid x_{<t})
 =Q_{\mathrm{coin}}(\mathord\cdot\mid x_{<t})
 =\operatorname{Bernoulli}(1/2).
\]
Show also that for every \(x_{<T}\in\{0,1\}^{T-1}\),
\[
 P_{\mathrm{par}}(\mathord\cdot\mid x_{<T})
 =\delta_{\operatorname{par}(x_{<T})},
 \qquad
 Q_{\mathrm{coin}}(\mathord\cdot\mid x_{<T})
 =\operatorname{Bernoulli}(1/2),
\]
where \(\operatorname{par}(x_{<T})=(\sum_{t<T}x_t)\bmod2\).

\item Use the KL chain rule to show that
\[
 \KL(P_{\mathrm{par}},Q_{\mathrm{coin}})=\log 2,
 \qquad
 \frac{\KL(P_{\mathrm{par}},Q_{\mathrm{coin}})}{T}
 =\frac{\log 2}{T}.
\]

\item Derive
\[
 \E_{P_{\mathrm{par}}}[c_{\mathrm{par}}]=0,
 \qquad
 \E_{Q_{\mathrm{coin}}}[c_{\mathrm{par}}]=\frac12,
\]
and verify the equality
\[
 \E_{Q_{\mathrm{coin}}}[c_{\mathrm{par}}]
 =1-e^{-\KL(P_{\mathrm{par}},Q_{\mathrm{coin}})}.
\]

\item \label{ex:positionwise-loss:last}
For this part, let $\cX$ be any finite alphabet,
and let $P,Q$ be probability distributions on
$\cX^+ = \cup_{t\ge 1} \cX^t$,
the set of finite sequences taking values
in $\cX$. Let $c:\cX^+\to [0,1]$.
Assume that $\E_P[c]=0$.
For $x\in \cX^+$, let $|x|$ denote its length.
Assume that \(P(\cX^T)>0\) and \(Q(\cX^T)>0\) for every \(T\ge1\).
For \(x\in\cX^T\), define the conditional probability mass functions
\[
 P_T(x)=\frac{P(x)}{P(\cX^T)},
 \qquad
 Q_T(x)=\frac{Q(x)}{Q(\cX^T)}.
\]
Show that \(\E_{P_T}[c]=0\) for every \(T\ge1\).
Prove or disprove the following statement:
If $\KL(P_T,Q_T)/T\to 0$ as $T\to\infty$ then $\lim_{T\to\infty}\E_{Q_T}[c]=0$.
That is, if the average KL between $P_T$ and $Q_T$ vanishes,
the cost under $Q_T$ vanishes.
\end{enumerate}
\end{exercisechunk}
