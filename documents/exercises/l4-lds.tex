\begin{exercisechunk}
\item \textbf{Observable recurrence.\suppmark}
\label[exercise]{ex:lds}
\exerciseprofile{2}{2}{2}{\exproof\quad\excalculation}{%
Use Cayley--Hamilton to derive an observable recurrence while showing that
hidden-state differences persist.}
Let \(S_{t+1}=AS_t\), where
\(A=\bigl(\begin{smallmatrix}\cos\theta&-\sin\theta\\
\sin\theta&\cos\theta\end{smallmatrix}\bigr)\), and let
\(Z_t=c^\top S_t\). Use Cayley--Hamilton to show that
\[
 Z_{t+2}=2\cos(\theta)Z_{t+1}-Z_t.
\]
Show that distinct initial states remain
at the same Euclidean distance under these dynamics.
\end{exercisechunk}
