\begin{exercisechunk}
\item \textbf{Chosen prompts and conditional support.\suppmark}
\label[exercise]{ex:chosen-prompts}
\exerciseprofile{2}{1}{1}{\exproof\quad\excalculation\quad\exconstruction\quad\exinterpretation}{%
Compare one deliberately chosen distinguishing prompt with the random
demonstrations needed to identify the correct rule.}
In \cref{prop:conditional-support}, choose a prompt on which
the two candidate rules disagree. Show that its expert label identifies
the correct rule. Compare the number of such queries with the number
of random demonstrations needed for failure probability at most
\(\delta\in(0,1/2)\).
\end{exercisechunk}
